13/// This function generates an [`Element`] that groups [`PluginPanels`] in a tabbed view.
14///
15/// Automatically organises launched plug-ins into clickable tabs, orchestrates tab removal,
16/// focus shifting, and manages custom fallback placeholder content when no tabs are active.
17///
18/// Can be used as resizable component in a [`PanelGroup`].
19///
20/// # Arguments
21/// * `position` - The [`Position`] variant determines some behavioural traits with respect to rendering.
22/// * `min_size` - The minimum size this element may be shrinked to.
23/// * `children` - Child elements to display in the empty panel body as placeholder text.
24#[component]
25pub fn TabbedGroup(
26    position: Position,
27    #[props(default = 0)] min_size: i32,
28    children: Element,
29) -> Element {
30    let uuid = use_signal(|| Uuid::new_v4());
31    let mut open_plugins = use_signal(|| TabbedPlugins::new());
32    let mut active_tab = use_signal(|| None);
33    let mut new_plugin = use_signal(|| None);
34
35    use_effect(move || {
36        if new_plugin().is_some() {
37            let id = open_plugins.write().insert(new_plugin.take().unwrap());
38            active_tab.set(Some(id));
39        }
40    });
41
42    let context: Option<GroupContext> = try_use_context();
43
44    let size = if let Some(context) = context {
45        context.children().read().get(&uuid.peek()).cloned()
46    } else {
47        None
48    };
49
50    rsx! {
51        div {
52            class: "panel tabbed",
53            flex_basis: if let Some(size) = size { "{size.size()}px" } else { "auto" },
54            onmounted: move |e: MountedEvent| async move { on_mounted(e, context, uuid(), min_size).await },
55            div {
56                class: "tabbed-header",
57                div {
58                    class: "tabbed-header-panel-group",
59                    for uuid in open_plugins() {
60                        PanelHeader {
61                            class: if active_tab.read().unwrap_or_default() == uuid { Some("tabbed-active".to_string()) } else { None },
62                            panel_name: open_plugins().get(&uuid).unwrap().to_string(),
63                            buttons: rsx! {
64                                CloseButton {
65                                    onclick: move |event: MouseEvent| {
66                                        event.stop_propagation();
67                                        if active_tab.peek().unwrap_or_default() == uuid {
68                                            let next = open_plugins.peek().find_next(&uuid);
69                                            active_tab.set(next);
70                                        }
71                                        open_plugins.write().remove(&uuid);
72                                    },
73                                }
74                            },
75                            onclick: move |_| active_tab.set(Some(uuid.clone())),
76                        },
77                    }
78                },
79                div {
80                    class: "panel-header-button-group",
81                    AddButton {
82                        position: position.clone(),
83                        opened_plugin: new_plugin,
84                    }
85                }
86            }
87            for tab in open_plugins() {
88                div {
89                    class: if active_tab().unwrap() == tab { "tabbed-body" } else { "tabbed-body hidden" },
90                    key: "{tab}",
91                    PluginPanel {
92                        headless: true,
93                        position: position.clone(),
94                        external_plugin: open_plugins().get(&tab).cloned(),
95                    }
96                }
97            }
98            if active_tab().is_none() {
99                div {
100                    class: "panel-body",
101                    { children },
102                },
103            }
104        }
105    }
106}

108/// Internal helper collection managing the sequential order and manifest data of open tabs.
109/// The mapping is performed using UUIDs, with each tab being assigned its own [`Uuid`].
110#[derive(PartialEq, Clone)]
111struct TabbedPlugins {
112    keys: Vec<Uuid>,
113    values: HashMap<Uuid, Manifest>,
114}
115
116impl TabbedPlugins {
117    /// Creates an empty collection of tabbed plug-ins.
118    pub fn new() -> Self {
119        Self {
120            keys: vec![],
121            values: HashMap::new(),
122        }
123    }
124
125    /// Inserts a plug-in manifest, assigning and returning a unique tab identifier ([`Uuid`]).
126    pub fn insert(&mut self, plugin: Manifest) -> Uuid {
127        let uuid = Uuid::new_v4();
128        self.keys.push(uuid);
129        self.values.insert(uuid, plugin);
130        uuid.clone()
131    }
132
133    /// Removes a plug-in from the collection based on its tab identifier.
134    pub fn remove(&mut self, uuid: &Uuid) {
135        self.keys.retain(|k| k != uuid);
136        self.values.remove(uuid);
137    }
138
139    /// Retrieves a reference to the plug-in manifest corresponding to a specific tab identifier.
140    pub fn get(&self, uuid: &Uuid) -> Option<&Manifest> {
141        self.values.get(uuid)
142    }
143
144    /// Determines which tab should gain focus after the specified tab is closed, based on the tab's
145    /// [`Uuid`] and returns the `Uuid` of the tab to gain focus next, if any is available.
146    pub fn find_next(&self, uuid: &Uuid) -> Option<Uuid> {
147        if let Some(index) = self.keys.iter().position(|u| u == uuid) {
148            if index == 0 {
149                return self.keys.get(1).cloned();
150            }
151            return self.keys.get(index - 1).cloned();
152        }
153        None
154    }
155
156    /// Retrieves a clone of the plug-in [`Manifest`] associated with the provided [`Uuid`].
157    /// If there is no `Manifest` found, None is returned.
158    pub fn filter(&self, uuid: &Uuid) -> Option<Manifest> {
159        self.values
160            .iter()
161            .filter_map(|(k, v)| if k == uuid { Some(v.clone()) } else { None })
162            .collect::<Vec<_>>()
163            .first()
164            .cloned()
165    }
166}
167
168impl IntoIterator for TabbedPlugins {
169    type Item = Uuid;
170    type IntoIter = IntoIter<Uuid>;
171
172    fn into_iter(self) -> Self::IntoIter {
173        self.keys.into_iter()
174    }
175}
