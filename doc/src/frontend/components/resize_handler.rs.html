8/// The UI component representing the resizable boundary bar.
9///
10/// Listens for mouse down and move events, computes the directional delta,
11/// and updates the layout tracking context to adjust flex-basis constraints, hence providing
12/// resizing for adjacent [`Panel`][crate::components::Panel]s.
13#[component]
14pub fn ResizeHandler() -> Element {
15    let mut resize_active = use_signal(|| false);
16    let mut last_mouse_position = use_signal(|| 0.);
17    let mut data = use_signal(|| PixelsRect::zero());
18
19    let context: GroupContext = use_context();
20
21    let orientation = context.orientation();
22    let orientation_trait = orientation.as_trait();
23    let own_range = use_memo(move || orientation_trait.extract_range(data()));
24
25    rsx! {
26        div {
27            class: "panel-resize-handler",
28            class: "{orientation}",
29            onmounted: move |e| async move {
30                data.set(
31                    e
32                    .get_client_rect()
33                    .await
34                    .unwrap_or(PixelsRect::zero())
35                );
36            },
37            onmousedown: move |e| {
38                resize_active.set(true);
39                last_mouse_position.set(orientation_trait.pointer_position(e));
40            },
41            if resize_active() {
42                div {
43                    class: "backdrop",
44                    onmousemove: move |e: MouseEvent| {
45                        let current_pos = orientation_trait.pointer_position(e);
46                        let last_pos = *last_mouse_position.peek();
47                        let delta = (current_pos - last_pos) as i32;
48
49                        let delta = resize(delta, own_range(), context);
50
51                        let translation = data.peek().translate(orientation_trait.translation_vector(delta as f64));
52
53                        data.set(translation);
54                        last_mouse_position.set(current_pos);
55                    },
56                    onmouseup: move |_| resize_active.set(false),
57                    onmouseleave: move |_| resize_active.set(false)
58                },
59            }
60        }
61    }
62}

64/// Computes the adjusted sizes for left/top and right/bottom sibling panels
65/// while strictly respecting specified minimum size bounds.
66///
67/// Returns the actual update the resizing operation performed.
68fn resize(delta: i32, own_range: Range<i32>, context: GroupContext) -> i32 {
69    let mut siblings = context.children();
70    let left_sibling = context.find_left_sibling(own_range.start);
71    let right_sibling = context.find_right_sibling(own_range.end);
72
73    let mut delta = delta;
74
75    if delta < 0 {
76        if let Some(ls) = left_sibling {
77            if let Some(ls) = siblings.write().get_mut(&ls) {
78                delta = ls.update_right(delta);
79            }
80        }
81        if let Some(rs) = right_sibling {
82            if let Some(rs) = siblings.write().get_mut(&rs) {
83                rs.update_left(delta);
84            }
85        }
86    } else {
87        if let Some(rs) = right_sibling {
88            if let Some(rs) = siblings.write().get_mut(&rs) {
89                delta = rs.update_left(delta);
90            }
91        }
92        if let Some(ls) = left_sibling {
93            if let Some(ls) = siblings.write().get_mut(&ls) {
94                ls.update_right(delta);
95            }
96        }
97    }
98    delta
99}
